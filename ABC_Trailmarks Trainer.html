class MonteCarloPredictor {

  constructor(iterations = 10000) {

    this.iterations = iterations;

  }


  // Generates randomized variance ε using Box-Muller transform

  _getEpsilon(variance = 0.1) {

    let u = 0, v = 0;

    while (u === 0) u = Math.random();

    while (v === 0) v = Math.random();

    let z = Math.sqrt(-2.0 * Math.log(u)) * Math.cos(2.0 * Math.PI * v);

    return z * variance;

  }


  // Extracts A (Accuracy), L (Latency), and W (Working Memory)

  _calculateMetrics(log, extra) {

    if (!log || log.length === 0) return { A: 0.5, L: 2000, W: 4 }; 

    const correct = log.filter(l => l.correct).length;

    const A = correct / log.length;

    const L = log.reduce((sum, l) => sum + (l.latencyMs || 2000), 0) / log.length;

    const W = extra?.memorylane?.maxSpan || 4;

    return { A, L, W };

  }


  runSimulation(log, extra, availableStations) {

    const { A, L, W } = this._calculateMetrics(log, extra);

    let stationScores = availableStations.map(st => ({ id: st.id, score: 0 }));


    for (let i = 0; i < this.iterations; i++) {

      stationScores.forEach(st => {

        // D simulates arbitrary difficulty variance per run (1.0 to 3.0)

        const D = 1.0 + (Math.random() * 2.0);

        const epsilon = this._getEpsilon();

        

        // Base formula: P(D) = (A * W) / (L * D) + ε

        // Latency is scaled down by 1000 for calculation balance

        const p = ((A * W * 1000) / (L * D)) + epsilon;

        st.score += p;

      });

    }


    return stationScores.sort((a, b) => b.score - a.score).map(st => st.id);

  }

}
